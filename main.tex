  \sectionhead{Visual overview}

  {\fontsize{35}{43}\selectfont\justifying
  Long-horizon prediction requires a world model that preserves useful information and
  decides how far ahead to reason. Temporal abstraction makes this tractable: latent
  steps span many timescales rather than a single tick. The Predictron learns a latent
  Markov reward process and blends predictions over several imagined depths, so the
  horizon is learned rather than fixed.}

  \vspace{0.45cm}

  \begin{columns}[t,totalwidth=\linewidth]
    \begin{column}{0.48\linewidth}
      \blockhead{Predictron}
      {\fontsize{31}{39}\selectfont\justifying
      $\lambda$-Preturn:
        \begin{equation}
          g^\lambda = \sum_{k=0}^{K} w_k g^k
        \end{equation}
        where:
        \begin{equation}
          w^k = \begin{cases} (1-\lambda^k) \prod_{j=0}^{k-1} \lambda^j & \text{if } k < K \\ \prod_{j=0}^{K-1} \lambda^j & \text{if } k = K \end{cases}
        \end{equation}

      \par}
    \end{column}
    \begin{column}{0.48\linewidth}
      \blockhead{Loss function}

      {\fontsize{31}{39}\selectfont\justifying
        \begin{equation}
          L^{0:K} = \frac{1}{2K} \sum_{k=0}^K \left\Vert{} \mathbb{E}_p[g \mid s] - \mathbb{E}_m[g^k \mid s] \right\Vert{}^2
        \end{equation}
          \begin{equation}
          L^\lambda = \frac{1}{2} \left\Vert{} \mathbb{E}_p[g \mid s] - \mathbb{E}_m[g^\lambda \mid s] \right\Vert{}^2
        \end{equation}
      \par}

    \end{column}
  \end{columns}

  \sectionhead{JESA: joint embedding successor architecture}

  \begin{columns}[t,totalwidth=\linewidth]
    \begin{column}{0.40\linewidth}
      {\fontsize{32}{40}\selectfont\justifying
      JESA trains the predictron against a \emph{separate} base encoder $\phi$ rather
      than against pixels: the model rolls its own latent forward one step and
      regresses onto $\phi(o_{t+1})$. The hatched arrow marks where the gradient is
      cut, which keeps the two branches from collapsing onto a constant
      embedding.\par}
    \end{column}

  \sectionhead{Kernels and reconstruction}

  {\fontsize{31}{39}\selectfont\justifying
  Written with a general kernel $k$, the return is
  \vspace{-0.2cm}
  \begin{equation*}
    G_k=\int_{0}^{\infty} k(t)\,r(t)\,\mathrm{d}t ,
  \end{equation*}
  so the familiar discounted return is the exponential choice $k(t)=e^{-t/\tau}$. How
  faithfully the event train reconstructs from a \emph{bank} of such returns depends
  on the kernel shape.\par}

  \sidehead{Why joint embedding?}

  \begin{center}
  \figslot{phi-embedding.png}{9}
  \end{center}
  \figcaption{Base-encoder embeddings $\phi(s)$ over a maze (hue $=$ embedding angle,
  value $=\lvert\phi\rvert$), with the PCA projection alongside.}

  \vspace{0.5cm}
  {\fontsize{27}{34}\selectfont\justifying
  $\phi$ is trained with \textbf{contrastive successor features} (CSF), the mutual
  information skill learning (MISL) objective of Zheng et al.~[2]. MISL maximises the
  empowerment-style mutual information between states and skills,\par}
  \vspace{0.1cm}
  {\fontsize{24}{30}\selectfont
  \begin{equation*}
    I(s;z)=\underbrace{H(z)}_{\text{skill diversity}}
           -\underbrace{H(z\mid s)}_{\text{identifiability}},
  \end{equation*}}
  \vspace{-0.2cm}
  {\fontsize{27}{34}\selectfont\justifying
  which organises the embedding by long-horizon reachability rather than by
  appearance.\par}

  \vspace{0.6cm}

  \sidehead{Successor features $\psi(s,z)$}
  \begin{center}
  \figslot{psi-successor-features.png}{14.5}
  \end{center}
  \figcaption{$\psi(s,z)$ per canonical $z$ (rows): maze colouring, PCA, rollouts.}
  \vspace{0.4cm}
  {\fontsize{27}{34}\selectfont\justifying
  Each $z$ induces its own field
  $\psi(s,z)=\mathbb{E}\big[\sum_k \gamma^k\phi(s_k)\mid s_0=s,z\big]$: a fixed $z$
  traces a coherent long-horizon path.\par}

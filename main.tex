  \begin{columns}[t,totalwidth=\linewidth]
    \begin{column}{0.48\linewidth}
      \blockhead{Predictron}
      {\fontsize{31}{39}\selectfont\justifying
      $k$-Step Preturn:
        \begin{equation}
          g^k = r^1 + \gamma^1 \left( r^2 + \gamma^2 \left( \dots + \gamma^{k-1}(r^k + \gamma^k v^k) \dots \right)\right)
        \end{equation}
      \vspace{0.5cm}

      $\lambda$-Preturn:
        \begin{equation}
          g^\lambda = \sum_{k=0}^{K} w_k g^k
        \end{equation}
        where:
        \begin{equation}
          w^k = \begin{cases} (1-\lambda^k) \prod_{j=0}^{k-1} \lambda^j & \text{if } k < K \\ \prod_{j=0}^{K-1} \lambda^j & \text{if } k = K \end{cases}
        \end{equation}
      \vspace{0.5cm}

      Backward Accumulation Sweep:
        \begin{equation}
          g^{k,\lambda} = (1-\lambda^k)v^k + \lambda^k \left( r^{k+1} + \gamma^{k+1} g^{k+1,\lambda} \right)
        \end{equation}
      \vspace{0.5cm}

      \par}
    \end{column}
    \begin{column}{0.48\linewidth}
      \blockhead{Loss function}

      {\fontsize{31}{39}\selectfont\justifying
        \begin{equation}
          L^{0:K} = \frac{1}{2K} \sum_{k=0}^K \left\Vert{} \mathbb{E}_p[g \mid s] - \mathbb{E}_m[g^k \mid s] \right\Vert{}^2
        \end{equation}
        \vspace{0.5cm}
        \begin{equation}
          L^\lambda = \frac{1}{2} \left\Vert{} \mathbb{E}_p[g \mid s] - \mathbb{E}_m[g^\lambda \mid s] \right\Vert{}^2
        \end{equation}
        \vspace{0.5cm}
        \begin{equation}
          L_{\text{cons}} = \frac{1}{2} \sum_{k=0}^K \left\Vert{} \mathbb{E}_m[g^\lambda \mid s] - \mathbb{E}_m[g^k \mid s] \right\Vert{}^2
        \end{equation}
      \par}

    \end{column}
  \end{columns}

  \sidehead{Markov reward process (MRP)}
  {\fontsize{29}{36}\selectfont
  An MRP is defined by:
  \begin{equation}
    s',\, r,\, \gamma = p(s,\alpha)
  \end{equation}
  \vspace{-0.35cm}
  \begin{posteritemize}
    \item $s'$: Next state
    \item $r$: Reward
    \item $\gamma$: Discount factor
  \end{posteritemize}}

  \vspace{0.8cm}

  Value function of an MRP $p$: the expected return from
state $s$
  \begin{equation}
    v_{p}(s) = E_{p} [g_{t} | s_{t} = s]
  \end{equation}
  
  \begin{equation}
    v_{p}(s) = E_{p} [r+\gamma v_{p}(s') | s]
  \end{equation}
